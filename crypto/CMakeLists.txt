cmake_minimum_required(VERSION 3.31)
project(rune_crypto CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenSSL REQUIRED)
add_library(rune_crypto STATIC src/Caps.cpp src/Aead.cpp src/TlsSession.cpp src/Kdf.cpp src/Scram.cpp)
target_include_directories(rune_crypto PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(rune_crypto PUBLIC OpenSSL::SSL OpenSSL::Crypto)
option(BUILD_TESTING "Build tests" OFF)
if(BUILD_TESTING)
    enable_testing()
    find_package(GTest REQUIRED)
    include(GoogleTest)
    add_executable(rune_crypto_tests tests/CapsTest.cpp tests/AeadTest.cpp tests/TlsSessionTest.cpp tests/CryptoRegressionTest.cpp tests/KdfTest.cpp tests/ScramTest.cpp)
    target_link_libraries(rune_crypto_tests PRIVATE rune_crypto GTest::gtest_main)
    # Test TLS fixtures (testdata/{server.key,server.pem,ca.pem}) are generated, never committed.
    include(${CMAKE_CURRENT_SOURCE_DIR}/cmake/GenerateTestCerts.cmake)
    rune_crypto_generate_test_certs("${CMAKE_CURRENT_BINARY_DIR}/testdata")
    gtest_discover_tests(rune_crypto_tests WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR})
endif()

option(BUILD_BENCHMARKS "Build benchmarks" OFF)
if(BUILD_BENCHMARKS)
    add_executable(rune_crypto_bench bench/AeadBench.cpp)
    target_link_libraries(rune_crypto_bench PRIVATE rune_crypto)
    target_compile_features(rune_crypto_bench PRIVATE cxx_std_23)
    find_package(benchmark REQUIRED)
    add_executable(rune_crypto_scram_bench bench/ScramBench.cpp)
    target_link_libraries(rune_crypto_scram_bench PRIVATE rune_crypto benchmark::benchmark)
    target_compile_features(rune_crypto_scram_bench PRIVATE cxx_std_23)
endif()
